<project_specification>
  <project_name>Hono Multi-Tenant SaaS Boilerplate</project_name>
  <version>1.0.0</version>
  <last_updated>2026-05-22</last_updated>

  <overview>
    Production-ready SaaS boilerplate for Cloudflare Workers with Hono.js,
    React, D1, KV, R2, and gateway-backed authentication through `@etus/auth`.
    The boilerplate owns product configuration, app-specific API routes,
    frontend integration, security middleware, and local developer helpers.
    `@etus/auth` owns OAuth, sessions, users, accounts, memberships,
    invitations, audit logging, and package-owned admin/account routes.
  </overview>

  <technology_stack>
    <runtime>
      <platform>Cloudflare Workers</platform>
      <database>D1</database>
      <sessions>KV Namespace</sessions>
      <storage>R2 Object Storage</storage>
    </runtime>

    <backend>
      <framework>Hono.js</framework>
      <auth_package>@etus/auth</auth_package>
      <validation>Zod</validation>
      <documentation>OpenAPI/Swagger for app-owned `/api/*` routes</documentation>
    </backend>

    <frontend>
      <framework>React 19</framework>
      <routing>TanStack Router</routing>
      <state>TanStack Query</state>
      <styling>Tailwind CSS</styling>
      <forms>React Hook Form + Zod</forms>
    </frontend>
  </technology_stack>

  <identity_and_auth>
    <gateway>ETUS OAuth Gateway (`ETUS_GATEWAY`, default `https://ag.etus.io`)</gateway>
    <sessions>HTTP-only cookie backed by KV and D1 `auth_sessions`</sessions>
    <login_flow>
      1. Browser opens `/auth/login`.
      2. `@etus/auth` redirects to the ETUS gateway.
      3. Gateway callback hits `/auth/callback`.
      4. `@etus/auth` provisions/updates `auth_users` and creates a session.
      5. Browser is redirected to the SPA route configured in `src/server/auth/setup.ts`.
    </login_flow>
    <frontend_contract>
      `useAuth()` calls `/auth/me` and expects `{ user: { id, email, name, picture, role } }`.
      Account-specific pages fetch `/accounts`, `/accounts/:id/members`, and
      `/accounts/:id/invitations` directly.
    </frontend_contract>
  </identity_and_auth>

  <roles_and_permissions>
    <product_roles>owner &gt; admin &gt; member &gt; guest</product_roles>
    <source_of_truth>src/server/auth/matrix.ts</source_of_truth>
    <account_membership_roles>admin | member | guest</account_membership_roles>
    <compatibility_note>
      `src/server/auth/package-compat.ts` validates membership role writes
      before `@etus/auth` account routes run, because `@etus/auth@0.3.0`
      still has hard-coded account-role behavior.
    </compatibility_note>
  </roles_and_permissions>

  <api_endpoints>
    <auth prefix="/auth">
      <endpoint method="GET" path="/login">Start ETUS gateway OAuth flow</endpoint>
      <endpoint method="GET" path="/callback">Handle OAuth callback and create session</endpoint>
      <endpoint method="POST" path="/logout">Destroy current session</endpoint>
      <endpoint method="GET" path="/me">Return current package user</endpoint>
      <endpoint method="POST" path="/test-login">Localhost-only E2E/dev session helper</endpoint>
    </auth>

    <admin_users prefix="/auth/admin">
      <endpoint method="GET" path="/users">List product users; requires package admin role</endpoint>
      <endpoint method="GET" path="/users/:id">Get product user</endpoint>
      <endpoint method="POST" path="/users/invite">Invite pending product user</endpoint>
      <endpoint method="POST" path="/users/:id/approve">Approve pending user</endpoint>
      <endpoint method="POST" path="/users/:id/deny">Deny pending user</endpoint>
      <endpoint method="PATCH" path="/users/:id">Update user role/status</endpoint>
      <endpoint method="DELETE" path="/users/:id">Delete user and invalidate sessions</endpoint>
    </admin_users>

    <accounts prefix="/accounts">
      <endpoint method="GET" path="/">List accounts for current user; returns `{ accounts }`</endpoint>
      <endpoint method="POST" path="/">Create account; creator membership role is `admin`</endpoint>
      <endpoint method="GET" path="/:id">Return `{ account, membership }`</endpoint>
      <endpoint method="PATCH" path="/:id">Update account; owner only</endpoint>
      <endpoint method="DELETE" path="/:id">Delete account; owner only</endpoint>
      <endpoint method="GET" path="/:id/members">List active members; returns `{ members }`</endpoint>
      <endpoint method="POST" path="/:id/members/invite">Create pending invitation; role required</endpoint>
      <endpoint method="PATCH" path="/:id/members/:userId">Update membership role</endpoint>
      <endpoint method="DELETE" path="/:id/members/:userId">Remove member</endpoint>
      <endpoint method="GET" path="/:id/invitations">List pending invitations</endpoint>
      <endpoint method="DELETE" path="/:id/invitations/:invitationId">Revoke invitation</endpoint>
    </accounts>

    <invitations prefix="/invitations">
      <endpoint method="POST" path="/:token/accept">Accept a pending invitation</endpoint>
    </invitations>

    <audit prefix="/audit">
      <endpoint method="GET" path="/logs">
        Query audit logs; supports `eventType`, `actorId`, `accountId`,
        `startDate`, `endDate`, `limit`, and `offset`.
      </endpoint>
      <endpoint method="POST" path="/cleanup">Delete logs past retention window</endpoint>
    </audit>

    <app_api prefix="/api">
      <endpoint method="GET" path="/doc">OpenAPI JSON for app-owned API routes</endpoint>
      <endpoint method="GET" path="/swagger">Swagger UI</endpoint>
      <endpoint method="POST" path="/storage/upload-url">Create internal upload URL</endpoint>
      <endpoint method="PUT" path="/storage/upload/:key">Upload file body to R2</endpoint>
      <endpoint method="GET" path="/storage/files/:key">Read file metadata/download target</endpoint>
      <endpoint method="DELETE" path="/storage/files/:key">Delete file from R2</endpoint>
    </app_api>
  </api_endpoints>

  <security>
    <csrf>
      State-changing browser requests require trusted Origin/Referer,
      and JSON content type for JSON APIs. The boilerplate intentionally does
      not send a decorative CSRF token header.
    </csrf>
    <headers>
      CSP, Trusted Types, HSTS, frame denial, nosniff, referrer policy, and
      restrictive permissions policy are applied by `securityHeaders()`.
    </headers>
    <app_api_auth>
      App-owned `/api/*` routes, except docs, run through `auth.middleware()`
      and `auth.accountMiddleware()`.
    </app_api_auth>
  </security>

  <environment_variables>
    <required>
      ENVIRONMENT
      APP_URL
      ETUS_GATEWAY
      ETUS_CLIENT_ID
      ETUS_ALLOWED_DOMAINS
      ETUS_ADMIN_EMAILS
    </required>
    <secrets>
      ETUS_CLIENT_SECRET
      SENDGRID_API_KEY
    </secrets>
    <optional>
      CORS_ORIGINS
      R2_PUBLIC_URL
      SENDGRID_FROM_EMAIL
    </optional>
  </environment_variables>

  <known_package_gaps>
    <gap>
      `@etus/auth@0.3.0` persists invitations but does not send invitation
      emails. The boilerplate currently creates/revokes invitations through the
      package routes; delivery should be implemented in `@etus/auth` or via a
      small package-supported hook before products depend on email delivery.
    </gap>
    <gap>
      Account membership role defaults are still hard-coded in the package.
      The boilerplate validates writes locally and should remove that adapter
      once `@etus/auth` exposes `multiTenant.roles/defaultRole`.
    </gap>
  </known_package_gaps>
</project_specification>
